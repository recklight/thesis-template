%% word in Chinese, the table-of-contents title back in English, each
%% changed on a different day.  Pick \ThesisLabelsChinese or
%% \ThesisLabelsEnglish below and every heading follows that choice.

%% ------------------------------------------------------------------
%% Blank-field test
%% ------------------------------------------------------------------
%% \ThesisIfBlank{\macro}{what to do when empty}{what to do otherwise}
%%
%% This is how an unused co-advisor slot disappears completely instead of
%% leaving an empty line on the cover.  The argument must expand to plain
%% text; that is true of every field in config/names.tex.
\newcommand{\ThesisIfBlank}[3]{%
  \edef\thesis@blanktest{#1}%
  \ifx\thesis@blanktest\@empty#2\else#3\fi}

%% \ThesisSpread{width}{text} -- stretch text across a fixed width.
%%
%% Used for the degree line on the cover, where the convention is to
%% space the characters out to a common width.
\newcommand{\ThesisSpread}[2]{\makebox[#1][s]{#2}}

%% ------------------------------------------------------------------
%% Symbol list
%% ------------------------------------------------------------------
%% For content/symbols.tex: a list whose label box has a fixed width, taken
%% from the widest symbol, so the descriptions line up in one column.  The
%% pattern is the one in The LaTeX Companion.
\newenvironment{SymEntry}[1]{%
  \begin{list}{}{%
    \settowidth{\labelwidth}{#1\hspace{1em}}%
    \setlength{\leftmargin}{\labelwidth}%
    \addtolength{\leftmargin}{\labelsep}%
    \setlength{\itemsep}{0.2ex}%
    \setlength{\parsep}{0pt}%
    \renewcommand{\makelabel}[1]{##1\hfill}}%
}{%
  \end{list}%
}
