%% thesis.tex -- main document.
%%
%% Compile it with:  thesis-cli build
%%
%% Do not run the TeX engine on any other file in this project.  Apart from
%% the spine (structure/coverbridge.tex, built with --target spine), every
%% other .tex file here is a fragment without a preamble, and compiling one
%% directly only produces a screenful of "Undefined control sequence".
%% 'thesis-cli build' refuses such a target for that reason.
%%
%% The engine is chosen in thesis.toml, not here.  config/cjk.tex asks the
%% iftex package which engine is running and loads the matching CJK setup.

\documentclass[12pt,a4paper]{thesis_report}

%% On the latex+dvipdfmx route pdfTeX writes DVI (\pdfoutput is 0), and
%% graphicx, xcolor and hyperref would then pick their dvips drivers:
%% graphicx cannot size a PNG that way, and hyperref's dvips links are not
%% what dvipdfmx reads.  Naming the dvipdfmx driver before the packages
%% load avoids both.  pdflatex and xelatex skip this block.
\usepackage{iftex}
\ifPDFTeX\ifnum\pdfoutput<1
  \PassOptionsToPackage{dvipdfmx}{graphicx}
  \PassOptionsToPackage{dvipdfmx}{xcolor}
  \PassOptionsToPackage{dvipdfmx}{hyperref}
\fi\fi
