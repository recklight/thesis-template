%% structure/frontpages.tex -- cover through list of symbols.
%%
%% Order: cover, title page, committee approval, authorisation, Chinese
%% abstract, English abstract, acknowledgements, contents, list of tables,
%% list of figures, list of symbols.  All of them are built; while drafting,
%% comment out this file's line in structure/chapters.tex to skip them.
%%
%% Optional fields (co-advisors, college, student number) are tested with
%% \ThesisIfBlank and take up no space when empty, so a thesis with one
%% advisor shows no gap where a second advisor would go.

%% ==================================================================
%% Cover
%% ==================================================================
\ThesisUnnumberedPage
\begin{center}
  {\setstretch{1.0}
   \LARGE\bfseries
   \univCname\\[0.6em]
   \ThesisIfBlank{\collCname}{}{\collCname\\[0.6em]}%
   \deptCname\\[0.6em]
   %% The degree line is conventionally set with wide inter-character
   %% spacing. \ThesisSpread stretches it to a fixed width; adjust the
   %% width if your degree name is longer than two characters.
   \ThesisSpread{4.5cm}{\degreeCname 論文}\\[3.5em]

   \Large
   \cTitle\\[1.5em]
   \large\mdseries
   \eTitle\\[4em]
  }

  {\setstretch{1.0}\large
   研究生：\myCname\\[0.8em]
   指導教授：\advisorCnameA\\[0.4em]
   \ThesisIfBlank{\advisorCnameB}{}{指導教授：\advisorCnameB\\[0.4em]}%
   \ThesisIfBlank{\advisorCnameC}{}{指導教授：\advisorCnameC\\[0.4em]}%
  }

  \vfill
  {\large 中華民國\ \cYear\ 年\ \cMonth\ 月}
\end{center}
\clearpage
